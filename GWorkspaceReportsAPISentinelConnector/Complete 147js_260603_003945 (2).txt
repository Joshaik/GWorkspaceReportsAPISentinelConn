// 4. VALIDATION LAYER: Parse and normalize raw binary inputs into defined qubit states
const validation = {
    validStates: new Set([TRUE_STATE, FALSE_STATE, ASYNC_STATE, SYNC_STATE]),

    parse: function(rawInput) {
        let cleaned = "";

        if (typeof rawInput === 'string') {
            cleaned = rawInput.replace(/[^01]/g, '').slice(-2).padStart(2, '0');
        } else if (typeof rawInput === 'number') {
            cleaned = (rawInput & 0b11).toString(2).padStart(2, '0');
        } else if (ArrayBuffer.isView(rawInput) || rawInput instanceof ArrayBuffer) {
            const view = ArrayBuffer.isView(rawInput) 
                ? new Uint8Array(rawInput.buffer, rawInput.byteOffset, rawInput.byteLength)
                : new Uint8Array(rawInput);
            
            if (view.length === 0) throw new RangeError("Empty binary buffer.");
            cleaned = (view[view.length - 1] & 0b11).toString(2).padStart(2, '0');
        } else {
            throw new TypeError(`Unsupported raw input type: ${typeof rawInput}`);
        }

        if (!this.validStates.has(cleaned)) {
            throw new RangeError(`Parsed state "${cleaned}" does not match any valid qubit definition.`);
        }

        return cleaned;
    },

    isValid: function(rawInput) {
        try {
            this.parse(rawInput);
            return true;
        } catch (e) {
            return false;
        }
    }
};

// 5. GATES LAYER: Classical Quantum Gate Approximations
const gates = {
    hadamard: function(state) {
        const hMap = {
            [TRUE_STATE]:  ASYNC_STATE, 
            [FALSE_STATE]: SYNC_STATE,  
            [ASYNC_STATE]: TRUE_STATE,  
            [SYNC_STATE]:  FALSE_STATE  
        };
        return hMap[state];
    },

    cnot: function(controlState, targetState) {
        if (controlState !== TRUE_STATE) {
            return targetState; 
        }

        const invertMap = {
            [TRUE_STATE]:  FALSE_STATE,
            [FALSE_STATE]: TRUE_STATE,
            [ASYNC_STATE]: SYNC_STATE,
            [SYNC_STATE]:  ASYNC_STATE
        };
        return invertMap[targetState];
    },

    applyToRegister: function(register, gateType, indexA, indexB) {
        const newRegister = [...register];
        
        if (gateType === 'H') {
            if (indexA < 0 || indexA >= newRegister.length) throw new RangeError("Target index out of bounds.");
            newRegister[indexA] = this.hadamard(newRegister[indexA]);
        } else if (gateType === 'CNOT') {
            if (indexB === undefined) throw new Error("CNOT requires a target index parameter.");
            if (indexA < 0 || indexA >= newRegister.length || indexB < 0 || indexB >= newRegister.length) {
                throw new RangeError("Control or Target register index out of bounds.");
            }
            newRegister[indexB] = this.cnot(newRegister[indexA], newRegister[indexB]);
        } else {
            throw new Error(`Unsupported gate operation: ${gateType}`);
        }
        
        return newRegister;
    }
};

// 6. PIPELINE LAYER: Unified parallel processing architecture supporting Multi-Qubits & Quadbits
const pipeline = {
    executeLogic: function(state, executableContext) {
        return logic[state].evaluate(executableContext);
    },

    executeResponse: async function(state, payload) {
        const targetResponse = response[state];
        const targetData = data[state];
        const resolution = await targetResponse.action(payload);

        return {
            state,
            status: targetResponse.status,
            message: targetResponse.message,
            telemetry: targetData.telemetry,
            resolution
        };
    },

    process: async function(rawInput, context) {
        try {
            const state = validation.parse(rawInput);
            const [logicResult, responseFrame] = await Promise.all([
                Promise.resolve(this.executeLogic(state, context)).catch(err => ({ error: err.message })),
                this.executeResponse(state, context).catch(err => ({ error: err.message }))
            ]);

            return {
                timestamp: Date.now(),
                success: true,
                qubit: {
                    state,
                    bitmask: data[state].bitmask,
                    properties: { ...data[state] }
                },
                execution: { logicFrame: logicResult, responseFrame: responseFrame }
            };
        } catch (error) {
            return { timestamp: Date.now(), success: false, error: error.message };
        }
    },

    processMany: async function(rawInputsArray, contexts) {
        try {
            if (!Array.isArray(rawInputsArray)) {
                throw new TypeError("Inputs must be provided within a structured execution array.");
            }

            const standardRegisterType = rawInputsArray.length === 4 ? "Quadbit" : `${rawInputsArray.length}-Qubit Register`;

            const operations = rawInputsArray.map(async (rawInput, index) => {
                const currentContext = Array.isArray(contexts) ? contexts[index] : contexts;
                const state = validation.parse(rawInput);

                const [logicResult, responseFrame] = await Promise.all([
                    Promise.resolve(this.executeLogic(state, currentContext)).catch(err => ({ error: err.message })),
                    this.executeResponse(state, currentContext).catch(err => ({ error: err.message }))
                ]);

                return {
                    registerIndex: index,
                    state,
                    bitmask: data[state].bitmask,
                    properties: { ...data[state] },
                    execution: { logicFrame: logicResult, responseFrame: responseFrame }
                };
            });

            const resolvedQubits = await Promise.all(operations);

            const combinedBitmask = resolvedQubits.reduce((acc, cur, idx) => acc | (cur.bitmask << (idx * 2)), 0);
            const totalEntropy = resolvedQubits.reduce((acc, cur) => acc + cur.properties.entropy, 0) / resolvedQubits.length;

            return {
                timestamp: Date.now(),
                success: true,
                topology: standardRegisterType,
                metrics: {
                    aggregateBitmask: combinedBitmask,
                    hexSignature: `0x${combinedBitmask.toString(16).toUpperCase()}`,
                    averageEntropy: parseFloat(totalEntropy.toFixed(4))
                },
                registers: resolvedQubits
            };

        } catch (error) {
            return {
                timestamp: Date.now(),
                success: false,
                error: error.message,
                topology: null,
                metrics: null,
                registers: []
            };
        }
    }
};
