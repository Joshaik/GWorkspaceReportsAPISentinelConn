import hashlib
import json
import time

def mine_aikhomu_block(block_data, difficulty_prefix="000"):
    """
    Increments a nonce until the block's SHA-256 hash starts with the required prefix.
    """
    nonce = 0
    print(f"Initializing Aikhomu Azure Blockchain Service...")
    print(f"Mining block... Target prefix: '{difficulty_prefix}'\n")
    
    while True:
        # Inject the current nonce into the block architecture
        block_data["nonce"] = nonce
        
        # Standardize the data structure into a strict, predictable string format
        serialized_data = json.dumps(block_data, sort_keys=True).encode('utf-8')
        
        # Generate the SHA-256 hexadecimal hash
        resulting_hash = hashlib.sha256(serialized_data).hexdigest()
        
        # Check if the block satisfies the proof of work condition
        if resulting_hash.startswith(difficulty_prefix):
            print("--- Block Successfully Mined ---")
            print(f"Final Hash: {resulting_hash}")
            print(f"Nonce Required: {nonce}")
            print("\nMined Block Payload Details:")
            print(json.dumps(block_data, indent=4))
            return block_data, resulting_hash
        
        nonce += 1
