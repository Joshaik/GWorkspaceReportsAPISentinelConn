import os
import json
import time
import hashlib
from azure.iot.device import IoTHubDeviceClient, MethodResponse, X509

DB_FILE = "aikhomu_blockchain.json"
HOSTNAME = "AikhomuHub.azure-devices.net"
DEVICE_ID = "aikhomu_edge_validator"
CERT_FILE = "aikhomu_device.crt"
KEY_FILE = "aikhomu_device.key"

def self_heal_database():
    """Wipes the corrupted chain, keeps valid baseline payloads, and remines the blocks."""
    print("⚠️ Self-healing sequence initiated locally...")
    if not os.path.exists(DB_FILE):
        return "Healing Failed: No database file found to recover."

    try:
        with open(DB_FILE, "r", encoding="utf-8") as f:
            corrupted_chain = json.load(f)
            
        healed_chain = []
        difficulty_prefix = "000"
        
        # Iterate through payloads, discard broken links, and reconstruct the block math sequentially
        for i, bad_block in enumerate(corrupted_chain):
            payload = bad_block.get("payload")
            previous_hash = "0" * 64 if i == 0 else healed_chain[-1]["block_hash"]
            
            block_structure = {
                "block_index": i,
                "timestamp": time.time(),
                "previous_hash": previous_hash,
                "payload": payload,
                "nonce": 0
            }
            
            # Re-mine the proof of work sequence to correct the data continuity
            while True:
                serialized = json.dumps(block_structure, sort_keys=True).encode('utf-8')
                current_hash = hashlib.sha256(serialized).hexdigest()
                if current_hash.startswith(difficulty_prefix):
                    break
                block_structure["nonce"] += 1
                
            completed_block = {
                "block_index": i,
                "timestamp": block_structure["timestamp"],
                "previous_hash": previous_hash,
                "nonce": block_structure["nonce"],
                "block_hash": current_hash,
                "payload": payload
            }
            healed_chain.append(completed_block)
            
        # Write healed data back onto the disk
        with open(DB_FILE, "w", encoding="utf-8") as f:
            json.dump(healed_chain, f, indent=4)
            
        print("✅ Database self-healing complete. Sequence restored.")
        return f"Success: Rebuilt and re-mined {len(healed_chain)} blocks safely."
    except Exception as e:
        return f"Critical failure during self-healing: {str(e)}"

def main():
    x509_auth = X509(cert_file=CERT_FILE, key_file=KEY_FILE)
    client = IoTHubDeviceClient.create_from_x509_certificate(hostname=HOSTNAME, device_id=DEVICE_ID, x509=x509_auth)
    client.connect()
    print("Aikhomu local system connected with self-healing listeners active.")

    def direct_method_handler(method_request):
        # Listen for the specific cloud trigger issued by the Azure Function
        if method_request.name == "repairLedger":
            repair_summary = self_heal_database()
            response_payload = {"status": "Healed", "details": repair_summary}
            status_code = 200
        else:
            response_payload = {"status": "Unhandled", "details": "Unknown method"}
            status_code = 404
            
        method_response = MethodResponse.create_from_method_request(method_request, status_code, response_payload)
        client.send_method_response(method_response)

    client.on_method_request_received = direct_method_handler
    
    try:
        while True: time.sleep(1)
    except KeyboardInterrupt:
        client.disconnect()

if __name__ == "__main__":
    main()
