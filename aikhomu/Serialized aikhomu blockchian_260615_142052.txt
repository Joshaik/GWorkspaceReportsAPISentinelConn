import hashlib
import json

def mine_aikhomu_block(block_data, difficulty_prefix="000"):
    nonce = 0
    while True:
        block_data["nonce"] = nonce
        serialized_data = json.dumps(block_data, sort_keys=True).encode('utf-8')
        resulting_hash = hashlib.sha256(serialized_data).hexdigest()
        if resulting_hash.startswith(difficulty_prefix):
            return nonce, resulting_hash
        nonce += 1
