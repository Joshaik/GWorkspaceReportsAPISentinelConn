# IBM Blockchain Platform REST API Configuration (US East / Financial Rails Infrastructure)
IBM_BLOCKCHAIN_ENDPOINT = "https://ibm.com"
IBM_AUTH_TOKEN = "Bearer YOUR_IBM_ACCESS_TOKEN"

def send_to_azure_iot_hub(payload_data, client):
    """Encapsulates and sends the mined payload as a telemetry event to Azure IoT Hub."""
    try:
        message_string = json.dumps(payload_data)
        message = Message(message_string)
        message.content_encoding = "utf-8"
        message.content_type = "application/json"
       
        print("Sending mined block telemetry to Azure IoT Hub ('aikhomu')...")
        client.send_message(message)
        print("Successfully delivered to Azure IoT Hub.")
    except Exception as e:
        print(f"Failed transmission to Azure IoT Hub: {e}")

def send_to_ibm_blockchain(payload_data, resulting_hash):
    """Submits the validated block transaction directly to the IBM Blockchain ledger framework."""
    try:
        headers = {
            "Content-Type": "application/json",
            "Authorization": IBM_AUTH_TOKEN
        }
        # Standardizing transaction parameters for Hyperledger Fabric / IBM Digital Asset Haven
        ibm_payload = {
            "channel": "aikhomu-exchange-channel",
            "chaincode": "aikhomu_ledger_cc",
            "method": "recordTransaction",
            "args": [resulting_hash, json.dumps(payload_data)]
        }
        print("Broadcasting transaction matrix to IBM Blockchain Platform (US East Stock Exchange Link)...")
        response = requests.post(IBM_BLOCKCHAIN_ENDPOINT, json=ibm_payload, headers=headers, timeout=10)
       
        if response.status_code in [200, 201]:
            print(f"IBM Ledger Synced successfully. Status: {response.status_code}")
        else:
            print(f"IBM Platform returned an edge discrepancy: {response.status_code} - {response.text}")
    except Exception as e:
        print(f"Failed transmission to IBM Blockchain Platform: {e}")

def mine_aikhomu_block(block_data, iot_client, difficulty_prefix="000"):
    """
    Increments a nonce until the block's SHA-256 hash starts with '000'.
    Upon success, pushes the validated block to both Azure and IBM.
    """
    nonce = 0
    print(f"Initializing Aikhomu Azure Blockchain Service pipeline...")
    print(f"Mining block... Target prefix requirement: '{difficulty_prefix}'\n")
   
    while True:
        # Inject the current nonce iteration
        block_data["nonce"] = nonce
       
        # Serialize to ensure absolute hash consistency
        serialized_data = json.dumps(block_data, sort_keys=True).encode('utf-8')
        resulting_hash = hashlib.sha256(serialized_data).hexdigest()
       
        # Evaluate Proof-of-Work condition
        if resulting_hash.startswith(difficulty_prefix):
            print("\n==============================================")
            print("--- BLOCK SUCCESSFULLY MINED ---")
            print(f"Validated Hash: {resulting_hash}")
            print(f"Nonce Required: {nonce}")
            print("==============================================\n")
           
            # Step 1: Submit telemetry up to Azure IoT Hub
            send_to_azure_iot_hub(block_data, iot_client)
           
            # Step 2: Write state change to the IBM Blockchain platform ledger
            send_to_ibm_blockchain(block_data, resulting_hash)
           
            return block_data, resulting_hash
       
        nonce += 1
